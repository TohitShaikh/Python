--------------------------------------------------------------------Sequence Category Data Type------------------------------------------------------------------------
-----------------------------------------------------------------------------bytearray--------------------------------------------------------------------------------
---------------
Example
---------------
>>> lst=[10,20,34,256,155,97]
>>> print(lst,type(lst)) ------->[10, 20, 34, 256, 155, 97] <class 'list'>
>>> ba=bytearray(lst) ------->ValueError: byte must be in range(0, 256)
>>> lst=[10,20,34,255,-155,97]
>>> print(lst,type(lst)) ------->[10, 20, 34, 255, -155, 97] <class 'list'>
>>> ba=bytearray(lst) ------->ValueError: byte must be in range(0, 256)	

>>> lst=[10,20,30,40,255,34]
>>> print(lst,type(lst)) ------->[10, 20, 30, 40, 255, 34] <class 'list'>
>>> ba=bytearray(lst)
>>> print(ba,type(ba)) ------->bytearray(b'\n\x14\x1e(\xff"') <class 'bytearray'>
>>> lst=[10,20,30,40,255,34]
>>> print(lst,type(lst)) ------->[10, 20, 30, 40, 255, 34] <class 'list'>
>>> ba=bytearray(lst)
>>> print(ba,type(ba)) ------->bytearray(b'\n\x14\x1e(\xff"') <class 'bytearray'>
>>> for val in ba:
...     print(val)
...
10
20
30
40
255
34
>>>
>>> ba[0] ------->10
>>> ba[1] ------->20
>>> ba[-1] ------->34
>>> ba[0:4] ------->bytearray(b'\n\x14\x1e(')
>>> for val in ba[0:4]:
...     print(val)
...
10
20
30
40
