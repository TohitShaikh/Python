---------------------------------------------------------------------list Category Data Type---------------------------------------------------------------------------
--------------------------------------------------------------------------list-----------------------------------------------------------------------------------------
----------------
Examples
----------------
>>> lst=[10,20,30,40,10,20,45,67,-56]
>>> print(lst,type(lst)) ----->[10, 20, 30, 40, 10, 20, 45, 67, -56] <class 'list'>
>>> lst1=[100,'Rossum',34.56,True,2+4j]
>>> print(lst1,type(lst1)) ----->[100, 'Rossum', 34.56, True, (2+4j)] <class 'list'>

Indexing and Slicing Operation---------------------------------------------------------------
>>> lst=[10,20,30,40,10,20,45,67,-56]
>>> print(lst,type(lst)) ----->[10, 20, 30, 40, 10, 20, 45, 67, -56] <class 'list'>
>>> lst1=[100,'Rossum',34.56,True,2+4j]
>>> print(lst1,type(lst1)) ----->[100, 'Rossum', 34.56, True, (2+4j)] <class 'list'>
>>>
>>> lst1=[100,'Rossum',34.56,True,2+4j]
>>> print(lst1,type(lst1),id(lst1)) ----->[100, 'Rossum', 34.56, True, (2+4j)] <class 'list'> 1363361527552
>>> lst1[0] ----->100
>>> lst1[-1] ----->(2+4j)
>>> lst1[0:4] ----->[100, 'Rossum', 34.56, True]
>>> lst1[0:3] ----->[100, 'Rossum', 34.56]
>>> lst1[::-1] ----->[(2+4j), True, 34.56, 'Rossum', 100]

>>> lst1=[100,'Rossum',34.56,True,2+4j]
>>> print(lst1,type(lst1),id(lst1)) ----->[100, 'Rossum', 34.56, True, (2+4j)] <class 'list'> 1363361446208
>>> lst1[0] ----->100 #Index Based Updation
>>> lst1[0]=150
>>> print(lst1,type(lst1),id(lst1)) ----->[150, 'Rossum', 34.56, True, (2+4j)] <class 'list'> 1363361446208
>>> lst1[1:3]=['Guido',55.55] #Slice Based Updation
>>> print(lst1,type(lst1),id(lst1)) ----->[150, 'Guido', 55.55, True, (2+4j)] <class 'list'> 1363361446208

Empty List----------------------------------------------------------------------------
>>> lst=[]
>>> print(lst,type(lst)) ----->[] <class 'list'>
>>> len(lst) ----->0

>>> lst2=list()
>>> print(lst2,type(lst2)) ----->[] <class 'list'>
>>> len(lst2) ----->0

Non-Empty List------------------------------------------------------------------------
>>> lst1=[10,'RS',23.45]
>>> print(lst1,type(lst1)) ----->[10, 'RS', 23.45] <class 'list'>
>>> len(lst1) ----->3

>>> s='RADAR'
>>> lst=list(s)
>>> print(lst,type(lst)) ----->['R', 'A', 'D', 'A', 'R'] <class 'list'>
>>>
>>> lst=list(range(10,20,2))
>>> print(lst,type(lst)) ----->[10, 12, 14, 16, 18] <class 'list'>

Special Point----------------------------------------------------------------------
>>> a=10
>>> print(a,type(a)) ----->10 <class 'int'>
>>> lst=list(a) ----->TypeError: 'int' object is not iterable

-----Alternative Solution--------
>>> a=10
>>> print(a,type(a)) ----->10 <class 'int'>
>>> lst=list([a])
>>> print(lst,type(lst)) ----->[10] <class 'list'>

>>> s='MISSISSIPPI'
>>> print(s,type(s)) ----->MISSISSIPPI <class 'str'>
>>> lst=list(s)
>>> print(lst,type(lst)) ----->['M', 'I', 'S', 'S', 'I', 'S', 'S', 'I', 'P', 'P', 'I'] <class 'list'>

Pre-Defined Function In list
--------------------------------------------------------------
help(list)

---------------------------------------------------
1] append()
---------------------------------------------------
>>> lst=[10,'RS']
>>> print(lst,type(lst),id(lst)) ----->[10, 'RS'] <class 'list'> 2887538887040
>>> lst.append('Python')
>>> print(lst,type(lst),id(lst)) ----->[10, 'RS', 'Python'] <class 'list'> 2887538887040
>>> lst.append(23.45)
>>> print(lst,type(lst),id(lst)) ----->[10, 'RS', 'Python', 23.45] <class 'list'> 2887538887040

>>> lst=[]
>>> print(lst,type(lst),id(lst)) ----->[] <class 'list'> 2887538885440
>>> len(lst) ----->0
>>> lst.append(10)
>>> lst.append(20)
>>> lst.append(10)
>>> lst.append('HYD')
>>> print(lst,type(lst),id(lst)) ----->[10, 20, 10, 'HYD'] <class 'list'> 2887538885440
>>> len(lst) ----->4

---------------------------------------------------
2] insert()
---------------------------------------------------

>>> lst=[10,20,30,40,50,10,60,70]
>>> print(lst,type(lst),id(lst)) ----->[10, 20, 30, 40, 50, 10, 60, 70] <class 'list'> 2887549159104
>>> lst.insert(2,'RS')
>>> print(lst,type(lst),id(lst)) ----->[10, 20, 'RS', 30, 40, 50, 10, 60, 70] <class 'list'> 2887549159104
>>> lst.insert(0,2+3j)
>>> print(lst,type(lst),id(lst)) ----->[(2+3j), 10, 20, 'RS', 30, 40, 50, 10, 60, 70] <class 'list'> 2887549159104

>>> lst=[10,20,30,40,50]
>>> print(lst,type(lst),id(lst)) ----->[10, 20, 30, 40, 50] <class 'list'> 2887549160128
>>> lst.insert(-1,155)
>>> print(lst,type(lst),id(lst)) ----->[10, 20, 30, 40, 155, 50] <class 'list'> 2887549160128
>>> lst.insert(-5,'RS')
>>> print(lst,type(lst),id(lst)) ----->[10, 'RS', 20, 30, 40, 155, 50] <class 'list'> 2887549160128

Special Point---------------------------------------------------------------
>>> lst=[10,'rs',34.56,'PYTHON']
>>> print(lst,type(lst),id(lst)) ----->[10, 'rs', 34.56, 'PYTHON'] <class 'list'> 2887549163840
>>> lst.insert(300,'HYD')
>>> print(lst,type(lst),id(lst)) ----->[10, 'rs', 34.56, 'PYTHON', 'HYD'] <class 'list'> 2887549163840
>>> print(lst,type(lst),id(lst)) ----->['BANG', 10, 'rs', 34.56, 'PYTHON', 'HYD'] <class 'list'> 2887549163840
