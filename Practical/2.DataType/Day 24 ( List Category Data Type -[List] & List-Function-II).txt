9] extend()
---------------------------------------------------
>>> lst1=[10,20,30,40]
>>> lst2=['RS','TR','DR']
>>> print(lst1,type(lst1),id(lst1)) -----> [10, 20, 30, 40] <class 'list'> 1728225505728
>>> print(lst2,type(lst2),id(lst2)) -----> ['RS', 'TR', 'DR'] <class 'list'> 1728225498368
>>> lst1.extend(lst2)
>>> print(lst1,type(lst1),id(lst1)) -----> [10, 20, 30, 40, 'RS', 'TR', 'DR'] <class 'list'> 1728225505728
>>> print(lst2,type(lst2),id(lst2)) -----> ['RS', 'TR', 'DR'] <class 'list'> 1728225498368

>>> lst1=[10,20,30,40]
>>> lst2=['RS','TR','DR']
>>> lst2.extend(lst1)
>>> print(lst2,type(lst2),id(lst2)) -----> ['RS', 'TR', 'DR', 10, 20, 30, 40] <class 'list'> 1728225497920
------------------------------OR-----------------------------------

>>> lst1=[10,20,30]
>>> lst2=['rs','tr','dr']
>>> lst3=[1.2,2.3]
>>> lst1.extend(lst2,lst3) -----> TypeError: list.extend() takes exactly one argument (2 given)
#TO Avoid The Above errow
>>> lst1.extend(lst2)
>>> lst1.extend(lst3)
>>> print(lst1) -----> [10, 20, 30, 'rs', 'tr', 'dr', 1.2, 2.3]

--------------------------------------------------------------
>>> lst1=[10,20,30]
>>> lst2=['rs','tr','dr']
>>> lst3=[1.2,2.3]
>>>
>>> lst1=lst1+lst2+lst3 #Most Important
>>> print(lst1) -----> [10, 20, 30, 'rs', 'tr', 'dr', 1.2, 2.3]
