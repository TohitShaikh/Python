---------------------------------------------------------------------list Category Data Type---------------------------------------------------------------------------
--------------------------------------------------------------------------tuple----------------------------------------------------------------------------------------
----------------
Examples
----------------
>>> t1=(10,20,30,40,50)
>>> print(t1,type(t1)) ----> (10, 20, 30, 40, 50) <class 'tuple'>

>>> t2=(10,'RS',True,2+3j,24.45)
>>> print(t2,type(t2)) ----> (10, 'RS', True, (2+3j), 24.45) <class 'tuple'>

Indexing & Slicing-----------------------------
>>> t2=(10,'RS',True,2+3j,24.45)
>>> print(t2,type(t2)) ----> (10, 'RS', True, (2+3j), 24.45) <class 'tuple'>
>>> t2[0] ----> 10
>>> t2[-1] ----> 24.45
>>> t2[1:4] ----> ('RS', True, (2+3j))
>>> t2[::-1] ----> (24.45, (2+3j), True, 'RS', 10)

Immutable-------------------------------------
>>> t2=(10,'RS',True,2+3j,24.45)
>>> print(t2,type(t2)) ----> (10, 'RS', True, (2+3j), 24.45) <class 'tuple'>
>>> t2[0]=100 ----> TypeError: 'tuple' object does not support item assignment

Empty tuple------------------------------------
>>> t1=()
>>> print(t1,type(t1),len(t1)) ----> () <class 'tuple'> 0
>>> t2=tuple()
>>> print(t2,type(t2),len(t2)) ----> () <class 'tuple'> 0

Non-Empty tuple-------------------------------
>>> t1=(10,20,30,10,40)
>>> print(t1,type(t1),len(t1)) ------> (10, 20, 30, 10, 40) <class 'tuple'> 5

Special Point-----------------
>>> lst=[10,'RS',34.90,'PYTHON']
>>> print(lst,type(lst)) ------> [10, 'RS', 34.9, 'PYTHON'] <class 'list'>
>>> t=tuple(lst)
>>> print(t,type(t)) ------> (10, 'RS', 34.9, 'PYTHON') <class 'tuple'>

>>> s='NISSAN'
>>> print(s,type(s)) ------> NISSAN <class 'str'>
>>> t=tuple(s)
>>> print(t,type(t)) ------> ('N', 'I', 'S', 'S', 'A', 'N') <class 'tuple'>
>>> lst=list(t)
>>> print(lst,type(lst)) ------> ['N', 'I', 'S', 'S', 'A', 'N'] <class 'list'>

>>> a=10
>>> print(a,type(a)) ------> 10 <class 'int'>
>>> a=(10)
>>> print(a,type(a)) ------> 10 <class 'int'>
>>> a=(10,)
>>> print(a,type(a)) ------> (10,) <class 'tuple'>
>>> a=10,
>>> print(a,type(a)) ------> (10,) <class 'tuple'>

Pre-Defined Function ---------------------------------------------------------------
>>> t1=(10,20,30,40,10,20,10,70)
>>> print(t1,type(t1)) ------> (10, 20, 30, 40, 10, 20, 10, 70) <class 'tuple'>
>>> t2.count(10) ------> 1
>>> t1.count(10) ------> 3
>>> t1.count(20) ------> 2
>>> t1.count(200) ------> 0


>>> for i,v in enumerate(t1):
...     if(v==10):
...         print(i,'--->',v)
...
0 ---> 10
4 ---> 10
6 ---> 10

Copy Function--------------
>>> print(t1,type(t1),id(t1)) ------> (10, 20, 30, 40, 10, 20, 10, 70) <class 'tuple'> 2142662820464
>>> t2=t1 #Deep Copy possible but not shallow
>>> print(t2,type(t2),id(t2)) ------> (10, 20, 30, 40, 10, 20, 10, 70) <class 'tuple'> 2142662820464

Sorted--------------------------
>>> t1=(10,-2,13,4,15,0,4,12)
>>> print(t1,type(t1),id(t1)) ------> (10, -2, 13, 4, 15, 0, 4, 12) <class 'tuple'> 1704369390192
>>> sorted(t1) ------> [-2, 0, 4, 4, 10, 12, 13, 15]
>>> print(t1,type(t1),id(t1)) ------> (10, -2, 13, 4, 15, 0, 4, 12) <class 'tuple'> 1704369390192
>>> x=sorted(t1)
>>> print(x,type(x),id(x)) ------> [-2, 0, 4, 4, 10, 12, 13, 15] <class 'list'> 1704370015104
>>> t1=tuple(x)
>>> print(t1,type(t1),id(t1)) ------> (-2, 0, 4, 4, 10, 12, 13, 15) <class 'tuple'> 1704369390976

>>> t1=(10,-2,13,4,15,0,4,12)
>>> print(t1,type(t1),id(t1)) ------> (10, -2, 13, 4, 15, 0, 4, 12) <class 'tuple'> 1704369390192
>>> t1=tuple(sorted(t1))
>>> print(t1,type(t1),id(t1)) ------> (-2, 0, 4, 4, 10, 12, 13, 15) <class 'tuple'> 1704369390976

>>> t1=(10,-2,13,4,15,0,4,12)
>>> print(t1,type(t1),id(t1)) ------> (10, -2, 13, 4, 15, 0, 4, 12) <class 'tuple'> 1704369390192
>>> t1=tuple(sorted(t1))
>>> t1=tuple(sorted(t1))[::-1]
>>> print(t1,type(t1),id(t1)) ------> (15, 13, 12, 10, 4, 4, 0, -2) <class 'tuple'> 1704369390528
>>> t1=tuple(sorted(t1)[::-1])
>>> print(t1,type(t1),id(t1)) ------> (15, 13, 12, 10, 4, 4, 0, -2) <class 'tuple'> 1704369390976

----------------------------------------------------------------Inner tuple/Nested tuple-----------------------------------------------------------------------------
---------------------------------------
Possibility-1 :- tuple in tuple
---------------------------------------
>>> t1=(10,'RS',(17,16,18),(77,78,98),"OUCET")
>>> print(t1,type(t1),id(t1)) ------> (10, 'RS', (17, 16, 18), (77, 78, 98), 'OUCET') <class 'tuple'> 3081075462336
>>> t1[0] ------> 10
>>> t1[1] ------> 'RS'
>>> t1[2] ------> (17, 16, 18)
>>> t1[2][1] ------> 16
>>> t1[2][-2] ------> 16

---------------------------------------
Possibility-2 :- list in tuple
---------------------------------------

>>> t1=(10,'RS',[17,16,18],[77,78,98],"OUCET")
>>> print(t1,type(t1),id(t1)) ------> (10, 'RS', [17, 16, 18], [77, 78, 98], 'OUCET') <class 'tuple'> 3081075843200
>>> t1[2].sort()
>>> print(t1,type(t1),id(t1)) ------> (10, 'RS', [16, 17, 18], [77, 78, 98], 'OUCET') <class 'tuple'> 3081075843200

---------------------------------------
Possibility-3 :- tuple in list
---------------------------------------
>>> t1=[10,'RS',(17,16,18),(77,78,98),"OUCET"]
>>> print(t1,type(t1),id(t1)) ------> [10, 'RS', (17, 16, 18), (77, 78, 98), 'OUCET'] <class 'list'> 3081076115840
>>> t1[0] ------> 10
>>> print(t1[2],type(t1[2])) ------> (17, 16, 18) <class 'tuple'>
