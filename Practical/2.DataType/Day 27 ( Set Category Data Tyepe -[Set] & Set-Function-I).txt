----------------------------------------------------------------------Set Category Data Type---------------------------------------------------------------------------
--------------------------------------------------------------------------set------------------------------------------------------------------------------------------
>>> s1={10,20,30,40,50,60,10}
>>> print(s1,type(s1)) ------> {50, 20, 40, 10, 60, 30} <class 'set'>
>>> print(s1,type(s1)) ------> {50, 20, 40, 10, 60, 30} <class 'set'>

>>> s2={10,'DR',34.56,True,2+3j,"Python"}
>>> print(s2,type(s2)) ------> {'DR', True, 34.56, 'Python', 10, (2+3j)} <class 'set'>
>>> print(s2,type(s2)) ------> {'DR', True, 34.56, 'Python', 10, (2+3j)} <class 'set'>

Indexing & Slicing----------------------------------------------------------------
>>> s1={10,20,30,40,50,60,10}
>>> print(s1,type(s1)) ------> {50, 20, 40, 10, 60, 30} <class 'set'>

>>> s1[1] ------> TypeError: 'set' object is not subscriptable
>>> s1[0:1] ------> TypeError: 'set' object is not subscriptable

Immutability-----------------------------------------------------------------------
>>> s1[0]=100 ------> TypeError: 'set' object does not support item assignment

Mutability--------------------------------------------------------------------------
>>> s1={10,20,30,40,50,60,10}
>>> print(s1,type(s1),id(s1)) ------> {50, 20, 40, 10, 60, 30} <class 'set'> 1768437211520
>>> s1.add('Python')
>>> print(s1,type(s1),id(s1)) ------> {50, 'Python', 20, 40, 10, 60, 30} <class 'set'> 1768437211520

Non-Empty Object----------------------------------------------------------------------
>>> lst=[10,20,30,40,10,20,30,40,50]
>>> print(lst,type(lst),len(lst)) ------> [10, 20, 30, 40, 10, 20, 30, 40, 50] <class 'list'> 9
>>> s=set(lst)
>>> print(s,type(s),len(s)) ------> {40, 10, 50, 20, 30} <class 'set'> 5

>>> s1='MISSIPPIPPI'
>>> print(s1,type(s1),len(s1)) ------> MISSIPPIPPI <class 'str'> 11
>>> s2=set(s1)
>>> print(s2,type(s2),len(s2)) ------> {'M', 'S', 'I', 'P'} <class 'set'> 4

>>> s2=set((10,10,10,20,20,20,30,30,30,40))
>>> print(s2,type(s2),id(s2)) ------> {40, 10, 20, 30} <class 'set'> 2180636565216

Special Point--------------------------------------------------------------------------
>>> a=10
>>> print(a,type(a)) ------> 10 <class 'int'>
>>> s=set(a) ------> TypeError: 'int' object is not iterable
>>> s=set([a])
>>> print(s,type(s)) ------> {10} <class 'set'>

>>> a=10
>>> print(a,type(a)) ------> 10 <class 'int'>
>>> s=set((a)) ------> TypeError: 'int' object is not iterable
>>> s=set((a,))
>>> print(s,type(s)) ------> {10} <class 'set'>

>>> x={10}
>>> print(x,type(x),len(x)) ------> {10} <class 'set'> 1

>>> s=set({True})
>>> print(s,type(s)) ------> {True} <class 'set'>

Pre-Defined Function In set

help(set)

---------------------------------------------------
1] add()
---------------------------------------------------
>>> s1={10,'RS'}

>>> s1.add(647.829)
>>> s1.add('Rossum')
>>> print(s1,type(s1),id(s1)) ------> {'Rossum', 'Python', 647.829} <class 'set'> 2180663944896

---------------------------------------------------
2] clear()
---------------------------------------------------
>>> s1={10,"RS"}
>>> print(s1,type(s1),id(s1)) ------> {10, 'RS'} <class 'set'> 2180663942880
>>> s1.clear()
>>> print(s1,type(s1),id(s1)) ------> set() <class 'set'> 2180663942880
>>> len(s1) ------> 0

>>> s1.clear()
>>> print(s1,type(s1),id(s1)) ------> set() <class 'set'> 2180663942880
>>> len(s1) ------> 0
>>> print(s1.clear()) ------> None
>>> print(set().clear()) ------> None
