----------------------------------------------------------------------Set Category Data Type---------------------------------------------------------------------------
--------------------------------------------------------------------------set------------------------------------------------------------------------------------------
---------------------------------------------------
9] union()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s3=s1.union(s2)
>>> print(s2,type(s2)) ------> {25, 20, 30, 15} <class 'set'>

>>> s4=s2.union(s1)
>>> print(s4,type(s4)) ------> {40, 10, 15, 20, 25, 30} <class 'set'>

---------------------------------------------------
10] intersection()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s3=s1.intersection(s2)
>>> print(s3,type(s3)) ------> {20, 30} <class 'set'>

>>> s4=s2.intersection(s1)
>>> print(s4,type(s4)) ------> {20, 30} <class 'set'>

---------------------------------------------------
11] difference()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s3=s1.difference(s2)
>>> print(s3,type(s3)) ------> {40, 10} <class 'set'>

>>> s4=s2.difference(s1)
>>> print(s4,type(s4)) ------> {25, 15} <class 'set'>

---------------------------------------------------
12] difference_update()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s3=s1.difference_update(s2)
>>> print(s3,type(s3)) ------> None <class 'NoneType'>
>>> print(s1,type(s1)) ------> {40, 10} <class 'set'>

>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s2.difference_update(s1)
>>> print(s2,type(s2)) ------> {25, 15} <class 'set'>

---------------------------------------------------
13] symmetric_difference()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s3=s1.symmetric_difference(s2)
>>> print(s3,type(s3)) ------> {40, 10, 15, 25} <class 'set'>

By Math Formula------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}
>>>
>>> s3=s1.union(s2).difference(s1.intersection(s2))
>>> print(s3,type(s3)) ------> {40, 25, 10, 15} <class 'set'>

---------------------------------------------------
14] symmetric_difference_update()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}

>>> s3=s1.symmetric_difference_update(s2)
>>> print(s3) ------> None
>>> print(s1) ------> {40, 10, 15, 25}

---------------------------------------------------
15] update()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2={15,20,25,30}
>>> s1.update(s2)
>>> print(s1) ------> {40, 10, 15, 20, 25, 30}

>>> s1={10,20}
>>> s2={'apple','mango'}
>>> s1.update(s2)
>>> print(s1) ------> {'apple', 10, 'mango', 20}

15] copy()
---------------------------------------------------
>>> s1={10,20,30,40}
>>> s2=s1.copy()
>>> print(s1,id(s1)) ------> {40, 10, 20, 30} 1659199184608
>>> print(s2,id(s2)) ------> {40, 10, 20, 30} 1659199187072
>>> s1.add(9) #It will perfrom shallow copy 
>>> s2.add('RS')
>>> print(s1,id(s1)) ------> {40, 9, 10, 20, 30} 1659199184608
>>> print(s2,id(s2)) ------> {40, 10, 'RS', 20, 30} 1659199187072
