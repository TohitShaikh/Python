Empty dict-------------------------------------------------------
>>> d1={}
>>> print(d1,type(d1)) ------> {} <class 'dict'>
>>> len(d1) ------> 0

>>> d2=dict()
>>> print(d2,type(d2)) ------> {} <class 'dict'>
>>> len(d2) ------> 0

Pre-Defined Function In list
-------------------------------------------------------------
help(dict)

---------------------------------------------------
1] pop()
---------------------------------------------------
>>> d1={100:1,200:2,300:3,400:4,500:5}
>>> print(d1,type(d1),id(d1)) ------> {100: 1, 200: 2, 300: 3, 400: 4, 500: 5} <class 'dict'> 2080794516608
>>> d1.pop(100) ------> 1
>>> print(d1,type(d1),id(d1)) ------> {200: 2, 300: 3, 400: 4, 500: 5} <class 'dict'> 2080794516608
>>> d1.pop(300) ------> 3
>>> print(d1,type(d1),id(d1)) ------> {200: 2, 400: 4, 500: 5} <class 'dict'> 2080794516608
>>> d1.pop(900) ------> KeyError: 900

>>> {}.pop(1) ------> KeyError: 1
>>> dict().pop(99) ------> KeyError: 99

---------------------------------------------------
2] popitem()
---------------------------------------------------

>>> d1={100:1,200:2,300:3,400:4,500:5}
>>> print(d1,type(d1),id(d1)) ------> {100: 1, 200: 2, 300: 3, 400: 4, 500: 5} <class 'dict'> 2080794631808
>>> d1.popitem() ------> (500, 5)
>>> print(d1,type(d1),id(d1)) ------> {100: 1, 200: 2, 300: 3, 400: 4} <class 'dict'> 2080794631808
>>> d1.popitem() ------> (400, 4)
>>> print(d1,type(d1),id(d1)) ------> {100: 1, 200: 2, 300: 3} <class 'dict'> 2080794631808
>>> d1.popitem() ------> (300, 3)
>>> print(d1,type(d1),id(d1)) ------> {100: 1, 200: 2} <class 'dict'> 2080794631808
>>> d1.popitem() ------> (200, 2)
>>> print(d1,type(d1),id(d1)) ------> {100: 1} <class 'dict'> 2080794631808
>>> d1.popitem() ------> (100, 1)
>>> print(d1,type(d1),id(d1)) ------> {} <class 'dict'> 2080794631808
>>> d1.popitem() ------> KeyError: 'popitem(): dictionary is empty'

>>> {}.popitem() ------> KeyError: 'popitem(): dictionary is empty'
>>> dict().popitem() ------> KeyError: 'popitem(): dictionary is empty'
