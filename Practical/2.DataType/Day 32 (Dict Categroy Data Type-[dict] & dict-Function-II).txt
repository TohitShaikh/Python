3] clear()
---------------------------------------------------
>>> d1={10:1.2,20:2.3,30:3.6,40:4.9,50:5.0}
>>> print(d1,type(d1),id(d1)) ------> {10: 1.2, 20: 2.3, 30: 3.6, 40: 4.9, 50: 5.0} <class 'dict'> 1691075566336
>>> len(d1) ------> 5
>>> d1.clear()
>>> print(d1,type(d1),id(d1)) ------> {} <class 'dict'> 1691075566336
>>> len(d1) ------> 0

>>> {}.clear() ------> Space as a result 
>>> print({}.clear()) ------> None
>>> dict().clear() ------> Space as a result
>>> print(dict().clear()) ------> None

4] copy()
---------------------------------------------------
>>> d1={10:1.2,20:2.3,30:3.6,40:4.9,50:5.0}
>>> print(d1,type(d1),id(d1)) ------> {10: 1.2, 20: 2.3, 30: 3.6, 40: 4.9, 50: 5.0} <class 'dict'> 1691075566656
>>> d2=d1.copy()
>>> print(d2,type(d2),id(d2)) ------> {10: 1.2, 20: 2.3, 30: 3.6, 40: 4.9, 50: 5.0} <class 'dict'> 1691075566208
>>> d1[10]=2.3 #Modification are idol to their object
>>> d2[10]=6.3
>>> print(d1,type(d1),id(d1)) ------> {10: 2.3, 20: 2.3, 30: 3.6, 40: 4.9, 50: 5.0} <class 'dict'> 1691075566656
>>> print(d2,type(d2),id(d2)) ------> {10: 6.3, 20: 2.3, 30: 3.6, 40: 4.9, 50: 5.0} <class 'dict'> 1691075566208

5] get()
---------------------------------------------------
>>> d1={10:1.2,20:2.3,30:3.6,40:4.9,50:5.0}
>>> print(d1) ------> {10: 1.2, 20: 2.3, 30: 3.6, 40: 4.9, 50: 5.0}
>>> bal=d1.get(10)
>>> print(bal) ------> 1.2
>>> d1.get(20) ------> 2.3
>>> bal=d1.get(30)
>>> print(bal) ------> 3.6
>>> bal=d1.get(101)
>>> print(bal) ------> None
>>> d1.get(120) ------> Space as a Result

9] update()
---------------------------------------------------
>>> d1={10:1.2,20:2.3}
>>> d2={'Python':1,'C':2}
>>> d1.update(d2)
>>> print(d1) -----> {10: 1.2, 20: 2.3, 'Python': 1, 'C': 2}
>>> d2.update(d1)
>>> print(d2) -----> {'Python': 1, 'C': 2, 10: 1.2, 20: 2.3}

>>> d1={10:1.2,20:2.3}
>>> d2={10:11.2,20:2.3}
>>> d1.update(d2)
>>> print(d1) -----> {10: 11.2, 20: 2.3}
