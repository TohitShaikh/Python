------------------------------------------------------------------------Assignment Operator----------------------------------------------------------------------------
Single Line Assignment----------------------------------
>>> a=10
>>> b=20
>>> c=a+b
>>> print(c,type(c)) ------> 30 <class 'int'>

>>> sno=10
>>> sname='Tohit'
>>> marks=34.57
>>> print(sno,sname,marks) ------> 10 Tohit 34.57

Multi Line Assignment----------------------------------
>>> a,b,c=10,20,30
>>> print(a,b,c) -----> 10 20 30

>>> age,name,sal=22,'Tohit',100.7
>>> print(age,name,sal) -----> 22 Tohit 100.7

>>> a,b=10,3
>>> c,d,e,f=a+b,a-b,a*b,a/b
>>> print(c) -----> 13
>>> print(d) -----> 7
>>> print(e) -----> 30
>>> print(f) -----> 3.3333333333333335

Swapping Logic2----------------------------------------
>>> a,b=10,30
>>> print(a,b) -----> 10 30

>>> a,b=b,a #Swaps Values Directly
>>> print(a,b) -----> 30 10

>>> a=100
>>> b=2000
>>> print(a,b) -----> 100 2000
>>> a,b=b,a
>>> print(a,b) -----> 
2000 100

Example-1

#Program for demonstrating rrelational operator
#RelationalOperatorEx1.py
a=int(input('Enter the value of a: '))
b=int(input('Enter The Value of b: '))
#Displaying the value
print('='*50)
print('\t Result of relational operator')
print('='*50)
print('\t\t {} > {} is {}'.format(a,b,a>b))
print('\t\t {} < {} is {}'.format(a,b,a<b))
print('\t\t {} == {} is {}'.format(a,b,a==b))
print('\t\t {} != {} is {}'.format(a,b,a!=b))
print('\t\t {} >= {} is {}'.format(a,b,a>=b))
print('\t\t {} <= {} is {}'.format(a,b,a<=b))
print('*'*50)

	Enter the value of a: -4
	Enter The Value of b: 5
	==================================================
	 Result of relational operator
	==================================================
		 -4 > 5 is False
		 -4 < 5 is True
		 -4 == 5 is False
		 -4 != 5 is True
		 -4 >= 5 is False
		 -4 <= 5 is True
	**************************************************

Special Point------------------------------
>>> x=10>2
>>> print(x,type(x)) -----> True <class 'bool'>
>>> x=float(10>2)
>>> print(x,type(x)) -----> 1.0 <class 'float'>
